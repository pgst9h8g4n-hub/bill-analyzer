## 项目结构
```
src/
├── lib/
│   ├── db/          # IndexedDB封装（expenses, stats, budget）
│   ├── export.ts    # CSV/JSON导出
│   ├── stores/      # auth store
│   └── utils/       # 格式化工具
├── routes/
│   ├── +page.svelte        # 首页（导航+欢迎+快捷操作）
│   ├── login/+page.svelte  # 登录/注册
│   ├── expenses/+page.svelte # 消费明细
│   ├── stats/+page.svelte   # 统计概览（ECharts）
│   ├── budget/+page.svelte  # 预算管理
│   ├── categories/+page.svelte # 分类管理
│   ├── settings/+page.svelte  # 账户设置
│   ├── export/+page.svelte    # 数据导出
│   └── ocr/+page.svelte       # OCR表单入口
└── static/
    ├── manifest.json  # PWA配置
    ├── sw.js         # Service Worker
    └── icons/        # 应用图标
```
